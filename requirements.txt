# Build-time dependency, deliberately held below 80.
# webrtcvad 2.0.10 has a legacy setup.py that imports `distutils`. That module
# was removed from the stdlib in Python 3.12, and setuptools 80 dropped the
# compatibility shim that used to stand in for it. On Python 3.12+ with
# setuptools >= 80, `pip3 install webrtcvad` fails to build with
# "ModuleNotFoundError: No module named 'distutils'". Pinning < 80 keeps the
# shim available so the wheel builds.
setuptools==79.0.1
